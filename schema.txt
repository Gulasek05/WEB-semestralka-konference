CREATE TABLE Role (
id INT AUTO_INCREMENT PRIMARY KEY,
nazev VARCHAR(50) NOT NULL UNIQUE)
ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE Uzivatel(
id INT  AUTO_INCREMENT PRIMARY KEY,
username VARCHAR(50) NOT NULL UNIQUE,
email VARCHAR(100) NOT NULL UNIQUE,
heslo_hash VARCHAR(255) NOT NULL,
role_id INT NOT NULL,
blocked TINYINT(1) NOT NULL DEFAULT 0,
vytvoreno TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (role_id) REFERENCES Role(id))
ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE Clanek(
id INT AUTO_INCREMENT PRIMARY KEY,
autor_id INT NOT NULL,
nazev VARCHAR(200) NOT NULL,
abstrakt TEXT NOT NULL,
soubor VARCHAR(255) NOT NULL,
stav ENUM('v_recenzi','publikovan','odmitnut') NOT NULL DEFAULT 'v_recenzi',
vytvoreno TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (autor_id) REFERENCES Uzivatel(id) ON DELETE CASCADE)
ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE Recenze (
id INT AUTO_INCREMENT PRIMARY KEY,
clanek_id INT NOT NULL,
recenzent_id INT NOT NULL,
originalita TINYINT NULL,
kvalita TINYINT NULL,
srozumitelnost TINYINT NULL,
komentar TEXT NULL,
upraveno TIMESTAMP NULL,
UNIQUE (clanek_id, recenzent_id),
FOREIGN KEY (clanek_id) REFERENCES Clanek(id) ON DELETE CASCADE,
FOREIGN KEY (recenzent_id) REFERENCES Uzivatel(id) ON DELETE CASCADE)
ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
